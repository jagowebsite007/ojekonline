/** * ============================================================================ * FILE 1: kode.gs * Deskripsi: Handler utama Web App (doPost & doGet) untuk menerima dan * menyimpan data pelanggan (Nama Pengguna, Alamat, No WA, Email) * dari aplikasi web ke Google Spreadsheet. * ============================================================================ */ /** * Fungsi doGet dijalankan saat URL Web App dibuka langsung melalui browser. * Berguna untuk memastikan koneksi Web App sudah aktif atau mengambil data. */ function doGet(e) { try { // Pastikan struktur sheet sudah siap if (typeof ensureDatabaseReady === 'function') { ensureDatabaseReady(); } var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'status'; // Daftar pelanggan if (action === 'list') { var customers = getAllCustomers(); return createJsonResponse({ status: 'success', total: customers.length, data: customers }); } // Daftar seluruh pesanan (digunakan oleh driver.html) if (action === 'list_orders') { var orders = getAllOrders(); return createJsonResponse({ status: 'success', total: orders.length, orders: orders }); } // Ambil detail 1 pesanan berdasarkan orderId (digunakan oleh index.html & driver.html) if (action === 'get_order') { var orderId = (e && e.parameter && e.parameter.orderId) ? e.parameter.orderId : ''; var order = getOrderById(orderId); return createJsonResponse({ status: order ? 'success' : 'not_found', order: order }); } // Daftar mitra driver if (action === 'list_drivers') { var drivers = getAllDrivers(); return createJsonResponse({ status: 'success', drivers: drivers }); } // Daftar riwayat penarikan saldo driver (opsional filter driverPhone) if (action === 'list_withdrawals') { var drvPhone = (e && e.parameter && e.parameter.driverPhone) ? e.parameter.driverPhone : ''; var withdrawals = getAllWithdrawals(drvPhone); return createJsonResponse({ status: 'success', total: withdrawals.length, withdrawals: withdrawals }); } // Respons default untuk pengecekan status URL Web App return createJsonResponse({ status: 'active', message: 'API Database Pelanggan & Driver aktif dan siap digunakan.', timestamp: new Date().toISOString() }); } catch (error) { return createJsonResponse({ status: 'error', message: error.toString() }); } } /** * Fungsi doPost dijalankan otomatis saat aplikasi pelanggan atau driver mengirim data. */ function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); if (typeof ensureDatabaseReady === 'function') { ensureDatabaseReady(); } if (!e || !e.postData || !e.postData.contents) { throw new Error('Tidak ada data JSON yang dikirimkan pada body request.'); } var payload = JSON.parse(e.postData.contents); var action = (payload.action || 'upsert_customer').toString().trim(); var timestamp = new Date(); var ss = SpreadsheetApp.getActiveSpreadsheet(); // --- ROUTE A: BUAT PESANAN BARU DARI PELANGGAN --- if (action === 'create_order') { var orderResult = createOrderRecord(ss.getSheetByName(DB_CONFIG.ORDER_SHEET_NAME), payload, timestamp); return createJsonResponse({ status: 'success', orderId: orderResult.orderId, message: 'Pesanan berhasil diteruskan ke radar driver.' }); } // --- ROUTE B: DRIVER MENERIMA ORDER --- if (action === 'accept_order') { var acceptResult = acceptOrderRecord(ss.getSheetByName(DB_CONFIG.ORDER_SHEET_NAME), payload); return createJsonResponse(acceptResult); } // --- ROUTE C: PEMBARUAN STATUS ORDER (JALAN / SELESAI / BATAL) --- if (action === 'update_order_status') { var statusResult = updateOrderStatusRecord( ss.getSheetByName(DB_CONFIG.ORDER_SHEET_NAME), ss.getSheetByName(DB_CONFIG.DRIVER_SHEET_NAME), payload ); return createJsonResponse(statusResult); } // --- ROUTE D: KIRIM PESAN CHAT PELANGGAN <-> DRIVER --- if (action === 'send_chat') { var chatResult = appendOrderChat(ss.getSheetByName(DB_CONFIG.ORDER_SHEET_NAME), payload); return createJsonResponse(chatResult); } // --- ROUTE E: SIMPAN / UPDATE STATUS MITRA DRIVER --- if (action === 'upsert_driver') { var driverResult = upsertDriverRecord(ss.getSheetByName(DB_CONFIG.DRIVER_SHEET_NAME), payload, timestamp); return createJsonResponse({ status: 'success', driver: driverResult }); } // --- ROUTE F: PENGAJUAN PENARIKAN SALDO KE REKENING DRIVER --- if (action === 'withdraw_driver') { var wdResult = createWithdrawalRecord( ss.getSheetByName(DB_CONFIG.WITHDRAW_SHEET_NAME), ss.getSheetByName(DB_CONFIG.DRIVER_SHEET_NAME), payload, timestamp ); return createJsonResponse(wdResult); } // --- ROUTE DEFAULT: SIMPAN / UPDATE DATA PELANGGAN --- var nama = (payload.nama || '').toString().trim(); var alamat = (payload.alamat || '').toString().trim(); var noWa = (payload.no_wa || '').toString().trim(); var email = (payload.email || '').toString().trim(); var aksi = (payload.aksi || 'Update Profil').toString().trim(); if (!nama && !noWa && !email) { throw new Error('Data pelanggan tidak lengkap (Nama, No WA, atau Email wajib diisi).'); } var customerSheet = ss.getSheetByName(DB_CONFIG.CUSTOMER_SHEET_NAME); var upsertResult = upsertCustomerRecord(customerSheet, { timestamp: timestamp, nama: nama, alamat: alamat, no_wa: noWa, email: email, aksi: aksi }); var logSheet = ss.getSheetByName(DB_CONFIG.LOG_SHEET_NAME); if (logSheet) { logSheet.appendRow([ timestamp, upsertResult.customerId, nama, alamat, "'" + noWa, email, aksi ]); } return createJsonResponse({ status: 'success', operation: upsertResult.operation, customerId: upsertResult.customerId, message: 'Data pelanggan berhasil disimpan ke Google Spreadsheet.', timestamp: timestamp.toISOString() }); } catch (error) { return createJsonResponse({ status: 'error', message: error.toString() }); } finally { lock.releaseLock(); } } /** * Menyimpan pelanggan baru atau memperbarui baris pelanggan lama * jika Email atau Nomor WA sudah terdaftar di sheet "Data_Pelanggan". */ function upsertCustomerRecord(sheet, record) { var lastRow = sheet.getLastRow(); var foundRowIndex = -1; var existingCustomerId = ''; var registeredTime = record.timestamp; // Cari apakah pelanggan sudah terdaftar berdasarkan No WA atau Email if (lastRow > 1) { var values = sheet.getRange(2, 1, lastRow - 1, 8).getValues(); var cleanTargetWa = normalizePhone(record.no_wa); var cleanTargetEmail = record.email.toLowerCase(); for (var i = 0; i < values.length; i++) { var rowId = values[i][0]; var rowRegTime = values[i][1]; var rowWa = normalizePhone(values[i][5]); var rowEmail = (values[i][6] || '').toString().toLowerCase().trim(); var matchWa = cleanTargetWa && rowWa && (cleanTargetWa === rowWa); var matchEmail = cleanTargetEmail && rowEmail && (cleanTargetEmail === rowEmail); if (matchWa || matchEmail) { foundRowIndex = i + 2; // +2 karena baris 1 adalah header dan array 0-indexed existingCustomerId = rowId; registeredTime = rowRegTime || record.timestamp; break; } } } // Jika pelanggan sudah ada, perbarui barisnya if (foundRowIndex !== -1) { sheet.getRange(foundRowIndex, 1, 1, 8).setValues([[ existingCustomerId, registeredTime, record.timestamp, record.nama, record.alamat, "'" + record.no_wa, record.email, record.aksi ]]); return { operation: 'updated', customerId: existingCustomerId, row: foundRowIndex }; } // Jika pelanggan baru, tambahkan baris baru dengan ID unik var newCustomerId = 'USR-' + Utilities.formatDate(record.timestamp, Session.getScriptTimeZone(), 'yyMMdd') + '-' + Math.floor(100 + Math.random() * 900); sheet.appendRow([ newCustomerId, record.timestamp, record.timestamp, record.nama, record.alamat, "'" + record.no_wa, record.email, record.aksi ]); return { operation: 'inserted', customerId: newCustomerId, row: sheet.getLastRow() }; } /** * Mengambil seluruh daftar pelanggan dari sheet "Data_Pelanggan" */ function getAllCustomers() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(DB_CONFIG.CUSTOMER_SHEET_NAME); if (!sheet || sheet.getLastRow() <= 1) { return []; } var data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 8).getValues(); var list = []; for (var i = 0; i < data.length; i++) { list.push({ id_pelanggan: data[i][0], waktu_terdaftar: data[i][1], terakhir_diperbarui: data[i][2], nama: data[i][3], alamat: data[i][4], no_wa: data[i][5], email: data[i][6], aksi_terakhir: data[i][7] }); } return list; } /** * ============================================================================ * FUNGSI PENARIKAN PENDAPATAN KE REKENING DRIVER * ============================================================================ */ function createWithdrawalRecord(withdrawSheet, driverSheet, payload, timestamp) { var amount = Number(payload.amount || 0); if (amount <= 0) { return { status: 'error', message: 'Nominal penarikan harus lebih dari 0.' }; } var withdrawId = payload.withdrawId || ('WD-' + Utilities.formatDate(timestamp, Session.getScriptTimeZone(), 'yyMMdd') + '-' + Math.floor(100 + Math.random() * 900)); var driverName = (payload.driverName || 'Mitra Driver').toString().trim(); var driverPhone = (payload.driverPhone || '-').toString().trim(); var bank = (payload.bank || 'BCA').toString().trim(); var accountNumber = (payload.accountNumber || '-').toString().trim(); var accountHolder = (payload.accountHolder || driverName).toString().trim(); withdrawSheet.appendRow([ withdrawId, timestamp, driverName, "'" + driverPhone, bank, "'" + accountNumber, accountHolder, amount, 'DIPROSES' ]); // Kurangi kolom Total Pendapatan pada sheet Data_Driver jika ditemukan if (driverSheet && driverSheet.getLastRow() > 1 && driverPhone) { var cleanPhone = normalizePhone(driverPhone); var values = driverSheet.getRange(2, 1, driverSheet.getLastRow() - 1, 8).getValues(); for (var i = 0; i < values.length; i++) { if (normalizePhone(values[i][3]) === cleanPhone) { var currentBal = Number(values[i][7] || 0); var nextBal = Math.max(0, currentBal - amount); driverSheet.getRange(i + 2, 8).setValue(nextBal); break; } } } return { status: 'success', withdrawId: withdrawId, message: 'Pengajuan penarikan saldo berhasil dicatat ke Spreadsheet.' }; } function getAllWithdrawals(driverPhoneFilter) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(DB_CONFIG.WITHDRAW_SHEET_NAME); if (!sheet || sheet.getLastRow() <= 1) return []; var data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).getValues(); var targetPhone = normalizePhone(driverPhoneFilter || ''); var list = []; for (var i = data.length - 1; i >= 0; i--) { var rowPhone = normalizePhone(data[i][3]); if (targetPhone && rowPhone !== targetPhone) continue; list.push({ id: data[i][0], time: data[i][1], driverName: data[i][2], driverPhone: data[i][3], bank: data[i][4], accountNumber: data[i][5], accountHolder: data[i][6], amount: Number(data[i][7] || 0), status: data[i][8] || 'DIPROSES' }); } return list; }
