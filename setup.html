/** * ============================================================================ * FILE 2: setup.gs * Deskripsi: Script inisialisasi untuk membuat struktur tabel, header kolom, * format warna, serta upgrade kolom GPS di Google Spreadsheet. * * CARA PENGGUNAAN: * 1. Buka Google Spreadsheet Anda -> klik menu "Ekstensi" -> "Apps Script". * 2. Pastikan terdapat 2 file: `Code.gs` (atau `Kode.gs`) dan `setup.gs`. * 3. Jalankan fungsi `setupDatabase()` dari file `setup.gs` ini satu kali * (klik tombol "Jalankan" / "Run" di atas editor). * 4. Klik "Terapkan" (Deploy) -> "Deployment baru" / "Kelola deployment" * -> pilih "Versi baru" -> Jenis: "Aplikasi Web" * -> Eksekusi sebagai: "Saya" -> Yang memiliki akses: "Siapa saja". * 5. Salin URL Web App (.../exec) yang sama persis ke Aplikasi Pelanggan * (index.html) dan Aplikasi Driver Anda. * ============================================================================ */ var DB_CONFIG = { CUSTOMER_SHEET_NAME: 'Data_Pelanggan', LOG_SHEET_NAME: 'Log_Aktivitas', ORDER_SHEET_NAME: 'Data_Pesanan', DRIVER_SHEET_NAME: 'Data_Driver', WITHDRAW_SHEET_NAME: 'Data_Penarikan' }; /** * Menyiapkan seluruh sheet database dan memformat header kolom secara otomatis. * Aman dijalankan ulang tanpa menghapus data yang sudah tersimpan. */ function setupDatabase() { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Setup Sheet Pelanggan: Data_Pelanggan var customerHeaders = [ 'ID Pelanggan', 'Waktu Terdaftar', 'Terakhir Diperbarui', 'Nama Pengguna', 'Alamat', 'No WA', 'Email', 'Aksi Terakhir' ]; createOrFormatSheet( ss, DB_CONFIG.CUSTOMER_SHEET_NAME, customerHeaders, [140, 150, 150, 170, 260, 140, 200, 180] ); // 2. Setup Sheet Log Aktivitas: Log_Aktivitas var logHeaders = [ 'Waktu', 'Tipe Aksi', 'Pelaku', 'Detail Aktivitas', 'Status' ]; createOrFormatSheet( ss, DB_CONFIG.LOG_SHEET_NAME, logHeaders, [150, 160, 160, 340, 100] ); // 3. Setup Sheet Pesanan Real-Time: Data_Pesanan (16 Kolom: Order + Chat + GPS + Google Maps) var orderHeaders = [ 'ID Order', 'Waktu Order', 'Layanan', 'Nama Pelanggan', 'No WA Pelanggan', 'Rute / Detail', 'Total Tarif', 'Metode Bayar', 'Status Order', 'Nama Driver', 'Plat Kendaraan', 'Pesan Chat', 'Koordinat Jemput', 'Koordinat Tujuan', 'Jarak (Km)', 'Link Google Maps' ]; createOrFormatSheet( ss, DB_CONFIG.ORDER_SHEET_NAME, orderHeaders, [120, 150, 110, 160, 140, 320, 120, 110, 160, 160, 160, 260, 175, 175, 100, 280] ); // 4. Setup Sheet Mitra Pengemudi: Data_Driver var driverHeaders = [ 'ID Driver', 'Terakhir Aktif', 'Nama Driver', 'No WA Driver', 'Jenis Kendaraan', 'Plat Kendaraan', 'Status Kerja', 'Total Pendapatan' ]; createOrFormatSheet( ss, DB_CONFIG.DRIVER_SHEET_NAME, driverHeaders, [140, 150, 170, 140, 130, 140, 120, 140] ); // 5. Setup Sheet Penarikan Saldo Driver: Data_Penarikan var withdrawHeaders = [ 'ID Penarikan', 'Waktu Pengajuan', 'Nama Driver', 'No WA Driver', 'Bank / E-Wallet', 'No Rekening', 'Atas Nama', 'Nominal Tarik', 'Status' ]; createOrFormatSheet( ss, DB_CONFIG.WITHDRAW_SHEET_NAME, withdrawHeaders, [140, 150, 160, 140, 140, 160, 160, 130, 110] ); SpreadsheetApp.flush(); Logger.log('Setup database selesai. Seluruh 5 sheet siap digunakan.'); } /** * Helper untuk membuat sheet baru (atau meng-upgrade header kolom jika sheet sudah ada) * serta mengatur gaya header dan lebar kolom. */ function createOrFormatSheet(ss, sheetName, headers, colWidths) { var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); } // Pastikan jumlah kolom mencukupi sebelum menulis header if (sheet.getMaxColumns() < headers.length) { sheet.insertColumnsAfter(sheet.getMaxColumns(), headers.length - sheet.getMaxColumns()); } var headerRange = sheet.getRange(1, 1, 1, headers.length); headerRange.setValues([headers]); headerRange .setBackground('#059669') .setFontColor('#ffffff') .setFontWeight('bold') .setHorizontalAlignment('center') .setVerticalAlignment('middle'); sheet.setRowHeight(1, 32); sheet.setFrozenRows(1); if (Array.isArray(colWidths)) { for (var i = 0; i < colWidths.length; i++) { sheet.setColumnWidth(i + 1, colWidths[i]); } } return sheet; } /** * Memastikan seluruh sheet dan 16 kolom GPS tersedia sebelum doGet / doPost berjalan. */ function ensureDatabaseReady() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var customerSheet = ss.getSheetByName(DB_CONFIG.CUSTOMER_SHEET_NAME); var logSheet = ss.getSheetByName(DB_CONFIG.LOG_SHEET_NAME); var orderSheet = ss.getSheetByName(DB_CONFIG.ORDER_SHEET_NAME); var driverSheet = ss.getSheetByName(DB_CONFIG.DRIVER_SHEET_NAME); var withdrawSheet = ss.getSheetByName(DB_CONFIG.WITHDRAW_SHEET_NAME); var needsOrderUpgrade = orderSheet && orderSheet.getLastColumn() < 16; if (!customerSheet || !logSheet || !orderSheet || !driverSheet || !withdrawSheet || needsOrderUpgrade) { setupDatabase(); } } /** * Utilitas opsional: membersihkan isi sheet Log_Aktivitas tanpa menghapus header. */ function clearActivityLogs() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var logSheet = ss.getSheetByName(DB_CONFIG.LOG_SHEET_NAME); if (logSheet && logSheet.getLastRow() > 1) { logSheet.getRange(2, 1, logSheet.getLastRow() - 1, logSheet.getLastColumn()).clearContent(); Logger.log('Log aktivitas berhasil dibersihkan.'); } }
